<Subtitle>A container element for rendering and cycling between multiple panels.</Subtitle>

A container element that renders a `dialog` element with `manual` popover behavior.  Modal's will capture and trap focus while open but will otherwise behave similaraly to ActiveContainer's and propagate events to its children.

### Examples

#### Dropdown Modal with 3 List Items

```jsx
const App = () => {
	const ref = useRef();

	const anchorName = useDashedIdent();

	const [isOpen, setIsOpen] = useState(false);
	const toggle = () => setIsOpen(!isOpen);

	return (
		<InputProvider inputRef={ ref }>

			<Button
				ref={ ref }
				onClick={ toggle }
				style={{ anchorName, }}
			> toggle </Button>

			<Modal
				node={ 'modal' }
				hasFocus={ isOpen }
				initial={ 'list' }
				anchorName={ anchorName }
				isOpen={ isOpen }
				setIsOpen={ setIsOpen }
			>
				<ActiveList node={ 'list' }>
					<ModalListItem node={ 0 }> Item { 0 } </ModalListItem>
					<ModalListItem node={ 1 }> Item { 1 } </ModalListItem>
					<ModalListItem node={ 2 }> Item { 2 } </ModalListItem>
				</ActiveList>
			</Modal>

		</InputProvider>
	)
}

export const ModalListItem = (props) => {
    const store = useContext(ModalContext);
    const onConfirm = () => {
        const { closeModal } = store.getState();
        closeModal();
    }

    return (
        <ActiveList.Item
            { ...props }
            onConfirm={ onConfirm }
        />
    )
}
```
